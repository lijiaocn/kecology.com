{%- comment -%} Resolve editorial layout from page.body_class or post categories. Usage: {% include editorial-layout.html part="class" %} {% include editorial-layout.html part="stylesheet" %} {%- endcomment -%} {%- assign editorial_class = page.body_class -%} {%- if editorial_class == nil or editorial_class == '' -%} {%- assign post_cats = page.categories | join: ',' | downcase -%} {%- if post_cats contains 'guides' -%} {%- assign editorial_class = 'guides-layout' -%} {%- elsif post_cats contains 'blog' -%} {%- assign editorial_class = 'blog-layout' -%} {%- endif -%} {%- endif -%} {%- if include.part == 'class' -%} {{- editorial_class -}} {%- elsif include.part == 'stylesheet' -%} {%- if editorial_class == 'home-layout' -%} {%- elsif editorial_class == 'blog-layout' -%} {%- elsif editorial_class == 'guides-layout' -%} {%- elsif editorial_class == 'tool-layout' -%} {%- endif -%} {%- endif -%}
