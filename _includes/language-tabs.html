{%- comment -%} Language tab bar for i18n list pages. Parameters: base_path — page base URL, e.g. "/", "/blog/", "/tool/" {%- endcomment -%} {%- assign default_lang = site.default_lang | default: "en" -%} {%- assign current_lang = page.lang | default: default_lang -%} {%- assign lang_data = site.data.i18n.index[current_lang] -%}
{%- for i in (0..site.languages.size) -%} {%- if i < site.languages.size -%} {%- assign lang_code = site.languages[i] -%} {%- assign lang_name = lang_data[lang_code] -%} {%- if lang_code == default_lang -%} {%- assign lang_url = include.base_path -%} {%- else -%} {%- assign lang_url = "/" | append: lang_code | append: include.base_path -%} {%- endif -%} {{ lang_name }} {%- endif -%} {%- endfor -%}
